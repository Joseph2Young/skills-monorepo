#获取多支期货
# panel=True时且pandas<0.25时返回[pandas.Panel]对象 , panel= False 或 pandas>=0.25时返回[pandas.DataFrame]对象
panel=get_price(['A2109.XDCE', 'A2111.XDCE'] ,count=2, panel=True,end_date='2020-12-19')
#返回[pandas.DataFrame]对象,行索引是[datetime.datetime]对象, 列索引是期货代号
df=panel['open']
print(df)
