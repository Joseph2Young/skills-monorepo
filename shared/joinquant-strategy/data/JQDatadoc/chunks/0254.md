```python
获取某只股票的资产负债表数据
# 查询'000001.XSHE'的资产负债表数据
q = query(
    balance
).filter(
    balance.code == '000001.XSHE'
)
# 运行时间2020-10-15
df = get_fundamentals(q,'2020-10-15')
# pubDate公司发布财报的日期,statDate财报统计的季度的最后一天,cash_equivalents货币资金
print(df[['pubDate','statDate','cash_equivalents']])
```
